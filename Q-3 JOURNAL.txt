#include<stdio.h>
#include<conio.h>
#include<string.h>
void main()
{
	int i,temp=0;
	clrscr();
	char name[5][20],search[20];
	
	printf("\n enter name:");
	for(i=0;i<5;i++)
	{
		scanf("%s",&name[i]);
	}
	printf("\n enter search element:\n");
	scanf("%s",&search);
	for(i=0;i<5;i++)
	{
		if(strcmp(name[i],search)==0)
		{
			temp=1;
			break;
		}
	}
	if(temp==1)
	{
		printf("\n search element found at position %d %s",i,search);
	}
	else
	{
		printf("\n not found");
	}
	getch():
}
